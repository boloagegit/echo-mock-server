package com.echo.config;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.beans.factory.ObjectProvider;
import com.echo.pipeline.HttpMockPipeline;
import com.echo.pipeline.MockRequest;
import com.echo.entity.Protocol;
import io.undertow.server.HttpServerExchange;
import io.undertow.util.Headers;
import java.nio.charset.StandardCharsets;
import java.util.HashMap;
import org.springframework.boot.web.embedded.jetty.JettyServletWebServerFactory;
import org.springframework.boot.web.embedded.tomcat.TomcatServletWebServerFactory;
import org.springframework.boot.web.embedded.undertow.UndertowServletWebServerFactory;
import org.springframework.boot.web.servlet.server.ServletWebServerFactory;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;

/** Temporary transport comparison; removed after selecting an implementation. */
@Configuration(proxyBeanMethods = false)
public class HttpTransportExperiment {
    @Bean
    ServletWebServerFactory experimentalWebServerFactory(
            @Value("${echo.http.transport:undertow}") String transport,
            ObjectProvider<HttpMockPipeline> pipeline) {
        return switch (transport) {
            case "jetty" -> {
                var factory = new JettyServletWebServerFactory();
                new org.springframework.boot.autoconfigure.websocket.servlet.JettyWebSocketServletWebServerCustomizer()
                        .customize(factory);
                yield factory;
            }
            case "tomcat" -> new TomcatServletWebServerFactory();
            default -> {
                var factory = new UndertowServletWebServerFactory();
                if (transport.startsWith("native-")) {
                    factory.addDeploymentInfoCustomizers(info -> info.addInitialHandlerChainWrapper(next -> exchange -> {
                        if (!exchange.getRequestURI().startsWith("/mock/")) { next.handleRequest(exchange); return; }
                        exchange.getRequestReceiver().receiveFullString((ex, body) -> {
                            Runnable action = () -> serve(ex, body.isEmpty() ? null : body, pipeline.getObject(), transport);
                            if (!transport.equals("native-inline")) ex.dispatch(action);
                            else action.run();
                        }, StandardCharsets.UTF_8);
                    }));
                }
                yield factory;
            }
        };
    }

    private static void serve(HttpServerExchange exchange, String body, HttpMockPipeline pipeline, String transport) {
        var headers = new HashMap<String, String>();
        exchange.getRequestHeaders().forEach(h -> headers.put(h.getHeaderName().toString(), h.getFirst()));
        String host = exchange.getRequestHeaders().getFirst("X-Original-Host");
        var request = MockRequest.builder().protocol(Protocol.HTTP)
                .method(exchange.getRequestMethod().toString()).path(exchange.getRequestURI().substring(5))
                .queryString(exchange.getQueryString().isEmpty() ? null : exchange.getQueryString())
                .headers(headers).body(body).targetHost(host == null || host.isBlank() ? "default" : host.trim())
                .clientIp(exchange.getSourceAddress().getAddress().getHostAddress()).build();
        var result = pipeline.execute(request);
        if (transport.equals("native-handoff")) {
            exchange.dispatch(exchange.getIoThread(), () -> write(exchange, result.getResponse()));
        } else {
            write(exchange, result.getResponse());
        }
    }

    private static void write(HttpServerExchange exchange, com.echo.pipeline.MockResponse response) {
        exchange.setStatusCode(response.getStatus());
        String value = response.getBody() == null ? "" : response.getBody();
        String trimmed = value.trim();
        exchange.getResponseHeaders().put(Headers.CONTENT_TYPE,
                trimmed.startsWith("{") || trimmed.startsWith("[") ? "application/json" :
                        trimmed.startsWith("<") ? "application/xml" : "text/plain;charset=UTF-8");
        exchange.getResponseSender().send(value, StandardCharsets.UTF_8);
    }
}
